'use client';

import React, { useEffect, useRef, useMemo, useState } from 'react';
import { Canvas } from '@react-three/fiber';
import { useFBX, useAnimations } from '@react-three/drei';
import { Group } from 'three';
import * as SkeletonUtils from 'three/examples/jsm/utils/SkeletonUtils.js';

export type ActionState =
  | 'idle'
  | 'lightAttack'
  | 'heavyAttack'
  | 'special'
  | 'block'
  | 'hit'
  | 'ko'
  | 'victory';

interface FighterModelProps {
  fighterId: 'void_lord_james' | 'chog_princess';
  action: ActionState;
  isPlayer2?: boolean;
}

const FIGHTER_FBX_MAP: Record<string, Record<ActionState, string>> = {
  void_lord_james: {
    idle: '/new fighters design/james/Mutant Flexing Muscles.fbx',
    lightAttack: '/new fighters design/james/Mutant Punch.fbx',
    heavyAttack: '/new fighters design/james/Punch To Elbow Combo.fbx',
    special: '/new fighters design/james/Double Leg Takedown - Attacker.fbx',
    block: '/new fighters design/james/Center Block.fbx',
    hit: '/new fighters design/james/Hit To Body.fbx',
    ko: '/new fighters design/james/Knocked Out.fbx',
    victory: '/new fighters design/james/Robot Hip Hop Dance.fbx',
  },
  chog_princess: {
    idle: '/new fighters design/princess/Mma Idle.fbx',
    lightAttack: '/new fighters design/princess/Combo Punch.fbx',
    heavyAttack: '/new fighters design/princess/Armada.fbx',
    special: '/new fighters design/princess/Medium Hit To Head.fbx',
    block: '/new fighters design/princess/Body Block.fbx',
    hit: '/new fighters design/princess/Light Hit To Head.fbx',
    ko: '/new fighters design/princess/Brutal Assassination.fbx',
    victory: '/new fighters design/princess/Victory Idle.fbx',
  },
};

function FighterNode({ fighterId, action, isPlayer2 = false }: FighterModelProps) {
  const group = useRef<Group>(null);
  const paths = FIGHTER_FBX_MAP[fighterId];

  const idleFbx = useFBX(paths.idle);
  const lightFbx = useFBX(paths.lightAttack);
  const heavyFbx = useFBX(paths.heavyAttack);
  const specialFbx = useFBX(paths.special);
  const blockFbx = useFBX(paths.block);
  const hitFbx = useFBX(paths.hit);
  const koFbx = useFBX(paths.ko);
  const victoryFbx = useFBX(paths.victory);

  const clonedScene = useMemo(() => SkeletonUtils.clone(idleFbx), [idleFbx]);

  const animations = useMemo(() => {
    const extractClip = (fbx: any, name: ActionState) => {
      if (!fbx.animations || fbx.animations.length === 0) return null;
      const clip = fbx.animations[0].clone();
      clip.name = name;
      return clip;
    };

    return [
      extractClip(idleFbx, 'idle'),
      extractClip(lightFbx, 'lightAttack'),
      extractClip(heavyFbx, 'heavyAttack'),
      extractClip(specialFbx, 'special'),
      extractClip(blockFbx, 'block'),
      extractClip(hitFbx, 'hit'),
      extractClip(koFbx, 'ko'),
      extractClip(victoryFbx, 'victory'),
    ].filter(Boolean) as any[];
  }, [idleFbx, lightFbx, heavyFbx, specialFbx, blockFbx, hitFbx, koFbx, victoryFbx]);

  const { actions } = useAnimations(animations, group);

  useEffect(() => {
    const currentAction = actions[action] || actions['idle'];
    if (!currentAction) return;

    Object.values(actions).forEach((act) => {
      if (act && act !== currentAction) act.fadeOut(0.15);
    });

    currentAction.reset().fadeIn(0.15).play();

    if (action === 'ko') {
      currentAction.setLoop(2200 as any, 1);
      currentAction.clampWhenFinished = true;
    }
  }, [action, actions]);

  return (
    <group
      ref={group}
      // P1 on the left (-1.8), P2 on the right (+1.8), facing each other
      position={[isPlayer2 ? 1.5 : -1.5, -1.6, 0]}
      rotation={[0, isPlayer2 ? -Math.PI / 3 : Math.PI / 3, 0]}
      scale={0.017}
    >
      <primitive object={clonedScene} />
    </group>
  );
}

interface BattleModalProps {
  isOpen: boolean;
  onClose: () => void;
  playerFighter?: 'void_lord_james' | 'chog_princess';
}

export default function BattleArenaModal({ isOpen, onClose, playerFighter = 'void_lord_james' }: BattleModalProps) {
  const [p1Action, setP1Action] = useState<ActionState>('idle');
  const [p2Action, setP2Action] = useState<ActionState>('idle');
  const [matchStatus, setMatchStatus] = useState<'fighting' | 'ko'>('fighting');

  // Simple automated combat sequence loop for testing
  useEffect(() => {
    if (!isOpen) return;
    const timer = setInterval(() => {
      const actionsList: ActionState[] = ['lightAttack', 'heavyAttack', 'special', 'block', 'hit'];
      const randomP1 = actionsList[Math.floor(Math.random() * actionsList.length)];
      const randomP2 = actionsList[Math.floor(Math.random() * actionsList.length)];

      setP1Action(randomP1);
      setP2Action(randomP2);

      setTimeout(() => {
        setP1Action('idle');
        setP2Action('idle');
      }, 1200);
    }, 3000);

    return () => clearInterval(timer);
  }, [isOpen]);

  if (!isOpen) return null;

  const opponentFighter = playerFighter === 'void_lord_james' ? 'chog_princess' : 'void_lord_james';

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/90 backdrop-blur-md p-4">
      <div className="relative w-full max-w-6xl h-[650px] bg-zinc-950 rounded-3xl border border-white/15 overflow-hidden flex flex-col shadow-2xl">
        
        {/* Top Health & Info HUD */}
        <div className="flex justify-between items-center px-8 py-4 bg-zinc-900/80 border-b border-white/10">
          <div className="flex items-center gap-4">
            <span className="font-bold text-red-500 uppercase tracking-widest text-sm">{playerFighter.replace('_', ' ')}</span>
            <div className="w-48 h-4 bg-zinc-800 rounded-full overflow-hidden border border-white/10">
              <div className="w-full h-full bg-red-600"></div>
            </div>
          </div>
          <div className="text-xl font-black text-amber-400 tracking-wider">VS</div>
          <div className="flex items-center gap-4 flex-row-reverse">
            <span className="font-bold text-purple-400 uppercase tracking-widest text-sm">{opponentFighter.replace('_', ' ')}</span>
            <div className="w-48 h-4 bg-zinc-800 rounded-full overflow-hidden border border-white/10">
              <div className="w-full h-full bg-purple-600"></div>
            </div>
          </div>
        </div>

        {/* 3D Battle Arena Canvas Stage (Zero flat background cards) */}
        <div className="relative flex-1 w-full bg-gradient-to-b from-zinc-950 via-zinc-900 to-black">
          <Canvas camera={{ position: [0, 0.4, 5], fov: 45 }}>
            <ambientLight intensity={2.0} />
            <directionalLight position={[5, 8, 5]} intensity={3.0} />
            <pointLight position={[-3, 2, 2]} intensity={1.5} color="#ec4899" />
            <pointLight position={[3, 2, 2]} intensity={1.5} color="#8b5cf6" />

            {/* Player 1 */}
            <FighterNode fighterId={playerFighter} action={p1Action} isPlayer2={false} />

            {/* Player 2 / Opponent */}
            <FighterNode fighterId={opponentFighter} action={p2Action} isPlayer2={true} />
          </Canvas>
        </div>

        {/* Bottom Control Bar */}
        <div className="flex justify-between items-center px-8 py-4 bg-zinc-900/90 border-t border-white/10">
          <span className="text-zinc-400 text-sm">3D Combat Engine Active</span>
          <button
            onClick={onClose}
            className="px-6 py-2 bg-red-600 hover:bg-red-700 text-white font-bold rounded-xl transition-all shadow-lg"
          >
            Exit Arena
          </button>
        </div>

      </div>
    </div>
  );
}
